function toggleCvCard(header) {
  var card = header.closest('.cv-card');
  var detail = card.querySelector('.cv-card-detail');
  if (!detail) return;
  var isOpen = detail.classList.contains('open');
  var allDetails = document.querySelectorAll('.cv-card-detail.open');
  for (var i = 0; i < allDetails.length; i++) {
    allDetails[i].classList.remove('open');
    allDetails[i].closest('.cv-card').querySelector('.cv-card-header').classList.remove('open');
  }
  if (!isOpen) {
    detail.classList.add('open');
    header.classList.add('open');
  }
}

function toggleCvTl(head) {
  var card = head.closest('.cv-tl-card');
  var detail = card.querySelector('.cv-tl-detail');
  if (!detail) return;
  var isOpen = detail.classList.contains('open');
  var allDetails = document.querySelectorAll('.cv-tl-detail.open');
  for (var i = 0; i < allDetails.length; i++) {
    allDetails[i].classList.remove('open');
    allDetails[i].closest('.cv-tl-card').querySelector('.cv-tl-head').classList.remove('open');
  }
  if (!isOpen) {
    detail.classList.add('open');
    head.classList.add('open');
  }
}
</script>
